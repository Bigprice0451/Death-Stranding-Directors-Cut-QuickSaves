cmake_minimum_required(VERSION 3.21)
project(DeathStrandingQuickSave VERSION 0.2.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
    message(FATAL_ERROR "Use Visual Studio x64. ReShade's overlay uses the Microsoft C++ ABI.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(quick_library STATIC src/library.cpp)
target_include_directories(quick_library PUBLIC src)
target_link_libraries(quick_library PUBLIC bcrypt)
add_library(diagnostics STATIC src/telemetry.cpp src/delivery_sample.cpp src/data_root.cpp)
target_link_libraries(diagnostics PUBLIC quick_library Shell32 Ole32 Uuid)
add_library(native_world STATIC src/native_image.cpp src/private_capture.cpp src/private_load.cpp)
target_link_libraries(native_world PUBLIC quick_library)
add_library(record STATIC src/thumbnail_image.cpp src/private_record.cpp)
target_link_libraries(record PUBLIC native_world windowscodecs ole32 diagnostics)
add_library(dlss_runtime STATIC src/dlss_runtime.cpp)
target_link_libraries(dlss_runtime PUBLIC Version)
add_executable(DLSSRuntime WIN32 src/dlss_install.cpp)
target_link_libraries(DLSSRuntime PRIVATE dlss_runtime diagnostics winhttp wintrust crypt32)
option(BUILD_RESHADE_ADDON "Build the in-game ReShade component" ON)
if(BUILD_RESHADE_ADDON)
    add_library(cinematic_hooks STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c

    target_include_directories(DirectorCutQuickSaves SYSTEM PRIVATE vendor/imgui vendor/reshade/include)

enable_testing()
function(quicksave_test name file lib)
    add_executable(${name} tests/${file}.cpp)
    target_link_libraries(${name} PRIVATE ${lib})
    add_test(NAME ${name} COMMAND ${name})
endfunction()
