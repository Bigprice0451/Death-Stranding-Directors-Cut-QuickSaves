function(quicksave_test name file lib)
    add_executable(${name} tests/${file}.cpp)
    target_link_libraries(${name} PRIVATE ${lib})
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# The add-on normally uses ReShade's ImGui function table. The input test
# links the same ImGui revision directly so no game or display is required.
set(IMGUI_TEST_SOURCE_DIR "" CACHE PATH "Optional ReShade 6.8.0 deps/imgui checkout")
if(IMGUI_TEST_SOURCE_DIR)
    add_library(browser_test_imgui STATIC)
    foreach(unit imgui imgui_draw imgui_tables imgui_widgets)
        target_sources(browser_test_imgui PRIVATE "${IMGUI_TEST_SOURCE_DIR}/${unit}.cpp")
    endforeach()
    target_include_directories(browser_test_imgui PUBLIC "${IMGUI_TEST_SOURCE_DIR}")
    target_compile_definitions(browser_test_imgui PUBLIC IMGUI_ENABLE_TEST_ENGINE)
    add_executable(browser_layout_test tests/browser_layout_test.cpp src/save_browser.cpp)
    target_include_directories(browser_layout_test PRIVATE src vendor/reshade/include)
    target_compile_options(browser_layout_test PRIVATE /FIreshade.hpp /utf-8)
    target_link_libraries(browser_layout_test PRIVATE browser_test_imgui imm32 dwmapi)
    add_test(NAME draggable_save_browser COMMAND browser_layout_test)
    add_executable(turning_panel_test tests/turning_panel_test.cpp src/turning_panel.cpp)
    target_include_directories(turning_panel_test PRIVATE src vendor/reshade/include)
    target_compile_definitions(turning_panel_test PRIVATE RESHADE_API_LIBRARY_EXPORT)
    target_compile_options(turning_panel_test PRIVATE /FIreshade.hpp /utf-8)
    target_link_libraries(turning_panel_test PRIVATE browser_test_imgui imm32 dwmapi)
    add_test(NAME turning_menu_preference COMMAND turning_panel_test)
endif()
